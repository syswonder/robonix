# Scene-container Python deps. Subset of pyproject.toml's
# [project.dependencies] minus what the base image already provides
# (numpy / cv-bridge / rclpy / tf2_ros).
#
# Two layers of deps:
#   1. Generic scene infrastructure (state, MCP, web).
#   2. Scene perception stack (per-frame detection +
#      segmentation + CLIP + 3D backproject + graph merge).
#
# Versions chosen for x86 + sm_120 (5090). Jetson port pins differ
# (NVIDIA's torch fork; mobile_sam + ViT-B-32 instead of larger
# variants; see notes in perception_concept_graphs.py).
#
# Heavy CV models load weights from /opt/models — the Dockerfile
# bakes those at build time. Never fetch at start (robonix invariant).

# ── Generic scene ────────────────────────────────────────────────
scipy>=1.11
mcp==1.29.0
# Runtime versions validated by scripts/run_python_codegen.sh. The generator
# remains isolated from the runtime environment.
grpcio==1.80.0
protobuf==6.33.6
uvicorn
httpx
Pillow
PyYAML
# Object persistence (warm restore across restarts). Embedded milvus-lite,
# its own scene-owned objects.db — same DB tech as system/memory's memsearch
# but a separate file/process. 2.6.10 is excluded (broken upsert); system/
# memory pulls pymilvus transitively (via memsearch), without an explicit pin.
# pymilvus 3.x no longer pulls milvus-lite through its old extra; pin the
# embedded backend explicitly so clean images retain warm persistence.
pymilvus!=2.6.10,>=2.5.0
milvus-lite==3.1.1

# ── Scene perception stack ───────────────────────────────────────
# torch 2.5+/cu124 because cu118 wheels don't run on sm_120 (Blackwell);
# ROS Humble's python3-numpy is pinned to 1.x so we let pip pick a
# compatible scipy build automatically.
opencv-python-headless>=4.8
ultralytics==8.4.115        # YOLO-World v2 + the SAM wrapper that
                            # accepts `.predict(image, bboxes=...)`
# YOLO-World prompt embeddings are frozen by docker/bake_yolo_world.py.
# clip-anytorch is a build-only dependency and must not ship at runtime.
open_clip_torch==2.32.0     # ViT-B-32 LAION-2B; small enough for
                            # Jetson, x86-5090 happily uses ViT-L/14
                            # by swapping the env var.
                            # Pinned <3 — open_clip 3.x changed the
                            # tokenizer / forward signature Scene relies on.
transformers==4.57.6        # tokenizers for CLIP text path; <5 because
                            # transformers 5.x renames model classes
                            # (Auto* import paths) and breaks CLIP loading.
safetensors==0.8.0          # tensor-only FP16 deployment checkpoints
huggingface-hub==0.36.2     # transformers/open_clip resolve weights via
                            # this; 1.x changed the cached_download API
                            # and breaks weight loading on first run.
